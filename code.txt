import heapq # liblary 
def huffman_encoding(characters, frequencies):
    heap = [] # array
    for i in range (len (characters)):
        heapq.heappush(heap, (frequencies[i],characters[i]))
    while len(heap) > 1:
        f1 , left = heapq.heappop(heap)
        f2 , right = heapq.heappop(heap)
        heapq.heappush(heap, (f1+f2,(left,right)))
    return heap[0][1]

                safe = False
            if board[i][col] == 1:
                pass

        for i in range (row):
            for j in range(n):
                if board[i][j] == 1:
                    if abs (row-i) == abs (col-j):
                        safe = False
